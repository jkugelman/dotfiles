# Zsh plugins, loaded by antidote from ~/.zshrc.
# Order matters here: zsh-syntax-highlighting must come last.

# Powerlevel10k prompt theme, on 256-color terminals only and never under Claude
# Code (see use-p10k in ~/.zshrc). Configured separately by ~/.config/zsh/p10k.zsh.

# Directory-stack navigation: Alt-Left/Right cycle through visited dirs like a
# browser's back/forward buttons. Keys are bound in
# ~/.config/zsh/rc.d/dir-navigation-keys.zsh.
ohmyzsh/ohmyzsh path:plugins/dircycle

# Fish-like autosuggestions from history and completions as you type.
zsh-users/zsh-autosuggestions

# Command-line syntax highlighting — catches typos before you hit enter.
# Deferred until just after the first prompt (was defer:2 under zplug).
zsh-users/zsh-syntax-highlighting kind:defer
